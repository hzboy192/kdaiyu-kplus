```java
@Service
public class UserProfileServiceImpl implements UserProfileService {

    @Autowired
    private UserProfileMapper userProfileMapper;

    @Autowired
    private SysUserMapper sysUserMapper;

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    @Override
    public UserProfileVO getUserProfile(Long userId) {
        // 先从缓存获取
        String cacheKey = "user:profile:" + userId;
        UserProfileVO cachedProfile = (UserProfileVO) redisTemplate.opsForValue().get(cacheKey);
        if (cachedProfile != null) {
            return cachedProfile;
        }

        // 查询数据库
        SysUserEntity user = sysUserMapper.selectOneById(userId);
        UserProfileEntity profile = userProfileMapper.selectOneByUserId(userId);

        // 构建VO对象
        UserProfileVO vo = new UserProfileVO();
        vo.setUserId(userId);
        vo.setUserName(user.getUserName());
        vo.setNickName(user.getNickName());
        vo.setAvatar(user.getAvatar());
        vo.setEmail(user.getEmail());
        vo.setPhonenumber(user.getPhonenumber());
        vo.setSex(user.getSex());

        if (profile != null) {
            vo.setIntroduction(profile.getIntroduction());
            vo.setTradingExperience(profile.getTradingExperience());
            vo.setInvestmentStyle(profile.getInvestmentStyle());
            vo.setRiskTolerance(profile.getRiskTolerance());
            vo.setFavoriteProducts(profile.getFavoriteProducts());
        }

        // 设置缓存
        redisTemplate.opsForValue().set(cacheKey, vo, 30, TimeUnit.MINUTES);

        return vo;
    }

    @Override
    public void updateUserProfile(Long userId, UserProfileUpdateDTO dto) {
        // 更新用户基本信息
        SysUserEntity user = new SysUserEntity();
        user.setUserId(userId);
        user.setNickName(dto.getNickName());
        user.setEmail(dto.getEmail());
        user.setPhonenumber(dto.getPhonenumber());
        user.setSex(dto.getSex());
        sysUserMapper.update(user);

        // 更新扩展信息
        UserProfileEntity profile = userProfileMapper.selectOneByUserId(userId);
        if (profile == null) {
            profile = new UserProfileEntity();
            profile.setUserId(userId);
            profile.setIntroduction(dto.getIntroduction());
            profile.setTradingExperience(dto.getTradingExperience());
            profile.setInvestmentStyle(dto.getInvestmentStyle());
            profile.setRiskTolerance(dto.getRiskTolerance());
            profile.setFavoriteProducts(dto.getFavoriteProducts());
            userProfileMapper.insert(profile);
        } else {
            profile.setIntroduction(dto.getIntroduction());
            profile.setTradingExperience(dto.getTradingExperience());
            profile.setInvestmentStyle(dto.getInvestmentStyle());
            profile.setRiskTolerance(dto.getRiskTolerance());
            profile.setFavoriteProducts(dto.getFavoriteProducts());
            userProfileMapper.update(profile);
        }

        // 清除缓存
        String cacheKey = "user:profile:" + userId;
        redisTemplate.delete(cacheKey);
    }
}
```

```java
@Service
public class TradeStatisticsServiceImpl implements TradeStatisticsService {

    @Autowired
    private UserTrainDtlMapper userTrainDtlMapper;

    @Autowired
    private UserTradeStatisticsMapper statisticsMapper;

    @Autowired
    private ProfitRankingMapper rankingMapper;

    @Autowired
    private RedisTemplate<String, Object> redisTemplate;

    @Override
    public TradeStatisticsVO getTradeStatistics(Long userId, String periodType,
                                                String startDate, String endDate) {

        // 构建缓存key
        String cacheKey = String.format("trade:stats:%s:%s:%s:%s",
            userId, periodType, startDate, endDate);

        // 尝试从缓存获取
        TradeStatisticsVO cachedStats = (TradeStatisticsVO) redisTemplate.opsForValue().get(cacheKey);
        if (cachedStats != null) {
            return cachedStats;
        }

        // 查询交易数据
        List<UserTrainDtlEntity> trades = userTrainDtlMapper.selectTradesByPeriod(
            userId, periodType, startDate, endDate);

        // 计算统计指标
        TradeStatisticsVO stats = calculateStatistics(trades);

        // 设置缓存
        redisTemplate.opsForValue().set(cacheKey, stats, 5, TimeUnit.MINUTES);

        return stats;
    }

    private TradeStatisticsVO calculateStatistics(List<UserTrainDtlEntity> trades) {
        TradeStatisticsVO stats = new TradeStatisticsVO();

        int totalTrades = trades.size();
        int winningTrades = 0;
        int losingTrades = 0;
        BigDecimal totalProfit = BigDecimal.ZERO;
        BigDecimal totalLoss = BigDecimal.ZERO;
        BigDecimal maxProfit = BigDecimal.ZERO;
        BigDecimal maxLoss = BigDecimal.ZERO;

        for (UserTrainDtlEntity trade : trades) {
            BigDecimal profit = trade.getProfit();
            if (profit.compareTo(BigDecimal.ZERO) > 0) {
                winningTrades++;
                totalProfit = totalProfit.add(profit);
                if (profit.compareTo(maxProfit) > 0) {
                    maxProfit = profit;
                }
            } else {
                losingTrades++;
                totalLoss = totalLoss.add(profit.abs());
                if (profit.abs().compareTo(maxLoss) > 0) {
                    maxLoss = profit.abs();
                }
            }
        }

        // 计算胜率
        BigDecimal winRate = totalTrades > 0 ?
            new BigDecimal(winningTrades).divide(new BigDecimal(totalTrades), 4, RoundingMode.HALF_UP)
                .multiply(new BigDecimal(100)) : BigDecimal.ZERO;

        // 计算净利润
        BigDecimal netProfit = totalProfit.subtract(totalLoss);

        // 计算盈利因子
        BigDecimal profitFactor = totalLoss.compareTo(BigDecimal.ZERO) > 0 ?
            totalProfit.divide(totalLoss, 2, RoundingMode.HALF_UP) : BigDecimal.ZERO;

        // 设置统计结果
        stats.setTotalTrades(totalTrades);
        stats.setWinningTrades(winningTrades);
        stats.setLosingTrades(losingTrades);
        stats.setWinRate(winRate);
        stats.setTotalProfit(totalProfit);
        stats.setTotalLoss(totalLoss);
        stats.setNetProfit(netProfit);
        stats.setProfitFactor(profitFactor);
        stats.setMaxProfit(maxProfit);
        stats.setMaxLoss(maxLoss);

        return stats;
    }
}
```

```java
     * 每月1日凌晨3点更新月榜
     */

        updateRanking("monthly");
    }

    private void updateRanking(String periodType) {
        // 获取所有活跃用户
        List<Long> activeUserIds = getActiveUserIds();

        // 计算每个用户的收益
        List<ProfitRankingEntity> rankings = new ArrayList<>();
        for (Long userId : activeUserIds) {
            TradeStatisticsVO stats = tradeStatisticsService.getTradeStatistics(
                userId, periodType, null, null);

            ProfitRankingEntity ranking = new ProfitRankingEntity();
            ranking.setUserId(userId);
            ranking.setPeriodType(periodType);
            ranking.setProfitAmount(stats.getNetProfit());
            ranking.setProfitRate(calculateProfitRate(userId, stats));
            ranking.setWinRate(stats.getWinRate());
            ranking.setTotalTrades(stats.getTotalTrades());
            ranking.setRankingDate(new Date());
```

### 1. Redis缓存配置
```java
@Configuration
public class RedisConfig {

    @Bean
    public RedisTemplate<String, Object> redisTemplate(RedisConnectionFactory factory) {
        RedisTemplate<String, Object> template = new RedisTemplate<>();
        template.setConnectionFactory(factory);

        // 使用Jackson2JsonRedisSerializer来序列化和反序列化redis的value值
        Jackson2JsonRedisSerializer<Object> serializer = new Jackson2JsonRedisSerializer<>(Object.class);
        ObjectMapper mapper = new ObjectMapper();
        mapper.setVisibility(PropertyAccessor.ALL, JsonAutoDetect.Visibility.ANY);
        mapper.activateDefaultTyping(mapper.getPolymorphicTypeValidator(),
            ObjectMapper.DefaultTyping.NON_FINAL);
        serializer.setObjectMapper(mapper);

        // 设置序列化方式
        template.setKeySerializer(new StringRedisSerializer());
        template.setValueSerializer(serializer);
        template.setHashKeySerializer(new StringRedisSerializer());
        template.setHashValueSerializer(serializer);
        template.afterPropertiesSet();

        return template;
    }

    @Bean
    public CacheManager cacheManager(RedisConnectionFactory factory) {
        RedisCacheConfiguration config = RedisCacheConfiguration.defaultCacheConfig()
            .entryTtl(Duration.ofMinutes(30))  // 默认缓存30分钟
            .serializeKeysWith(RedisSerializationContext.SerializationPair
                .fromSerializer(new StringRedisSerializer()))
            .serializeValuesWith(RedisSerializationContext.SerializationPair
                .fromSerializer(new GenericJackson2JsonRedisSerializer()))
            .disableCachingNullValues();

        // 设置不同缓存的过期时间
        Map<String, RedisCacheConfiguration> cacheConfigurations = new HashMap<>();
        cacheConfigurations.put("userProfile", config.entryTtl(Duration.ofMinutes(30)));
        cacheConfigurations.put("tradeStatistics", config.entryTtl(Duration.ofMinutes(5)));
        cacheConfigurations.put("profitRanking", config.entryTtl(Duration.ofMinutes(10)));
        cacheConfigurations.put("communityPosts", config.entryTtl(Duration.ofMinutes(15)));

        return RedisCacheManager.builder(factory)
            .cacheDefaults(config)
            .withInitialCacheConfigurations(cacheConfigurations)
            .build();
    }
}
```

```java
                    "/api/user/preferences",
                    "/api/trade/statistics",
                    "/api/trade/my-ranking"
                ).authenticated()

                // 需要特定权限的接口

                .antMatchers("/api/community/posts/**").hasAnyAuthority("COMMUNITY_WRITE")
                .antMatchers("/api/membership/**").hasAnyAuthority("MEMBERSHIP")

                // 其他接口需要认证
                .anyRequest().authenticated()
            .and()

            // 添加JWT过滤器
            .addFilterBefore(jwtAuthenticationTokenFilter(),
                UsernamePasswordAuthenticationFilter.class)

            // 异常处理
            .exceptionHandling()
                .authenticationEntryPoint(authenticationEntryPoint())
                .accessDeniedHandler(accessDeniedHandler());
    }
```

### 2. 数据权限控制

```java
@Component
public class DataPermissionAspect {

    @Pointcut("@annotation(cn.kdaiy.common.boot.annotation.DataScope)")
    public void dataScopePointCut() {
    }

    @Around("dataScopePointCut()")
    public Object around(ProceedingJoinPoint point) throws Throwable {
        // 获取当前用户
        LoginUser loginUser = SecurityUtils.getLoginUser();
        if (loginUser == null) {
            throw new ServiceException("用户未登录");
        }

        // 获取数据权限
        DataScope dataScope = getDataScope(loginUser);

        // 设置数据权限
        handleDataScope(point, dataScope);

        return point.proceed();
    }

    private void handleDataScope(ProceedingJoinPoint joinPoint, DataScope dataScope) {
        Object[] args = joinPoint.getArgs();
        for (Object arg : args) {
            if (arg instanceof BaseEntity) {
                BaseEntity baseEntity = (BaseEntity) arg;
                baseEntity.getParams().put("dataScope", dataScope);
            }
        }
    }
}
```
